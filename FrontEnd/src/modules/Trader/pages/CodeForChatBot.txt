import React, { useEffect, useMemo, useState } from "react";
import { useTradingRunManagement } from "../../hooks/useTradingRunManagement";

  CreateTradingRunCommand,
  UpdateTradingRunCommand,
  TradingRunInfoView,
  TradingRunDetailView,
  StrategyInfoView,
} from "../../models";

  TradingRunState,
  TRADING_RUN_STATE_LABELS,  
  TIMEFRAME_LABELS,
  TimeFrame,

  ExecutionMode,

  EXECUTION_MODE_COLORS,

const ExecutionModeBadge: React.FC<{ mode: number }> = ({ mode }) => {
  const m = mode as ExecutionMode;
  const label = EXECUTION_MODE_LABELS[m] ?? "—";
  const colors = EXECUTION_MODE_COLORS[m] ?? "bg-slate-700 text-slate-300";

  return (
    <span
      className={`inline-flex rounded-full px-2 py-0.5 text-[10px] font-medium whitespace-nowrap ${colors}`}
    >
      {m === ExecutionMode.Paper ? "📄 " : "🔴 "}
      {label}
    </span>
  );
};

export const TradingRunsManagementPage: React.FC = () => {
  const {
    runs,
    loading,
    busyId,
    error,
    refresh,
    handleCreate,
    handleUpdate,
    handleStart,
    handleStop,
    handleDelete,
  } = useTradingRunManagement();

  const [strategies, setStrategies] = useState<StrategyInfoView[]>([]);
  const [formOpen, setFormOpen] = useState(false);
  const [formMode, setFormMode] = useState<"create" | "edit">("create");
  const [formDetail, setFormDetail] =
    useState<TradingRunDetailView | null>(null);
  const [formSaving, setFormSaving] = useState(false);

  const [stopTarget, setStopTarget] =
    useState<TradingRunInfoView | null>(null);
  const [stopSaving, setStopSaving] = useState(false);

  const [deleteTarget, setDeleteTarget] =
    useState<TradingRunInfoView | null>(null);
  const [deleteSaving, setDeleteSaving] = useState(false);

  const [search, setSearch] = useState("");
  const [stateFilter, setStateFilter] = useState<string>("");

  /* ─── Load strategies once ─── */
  useEffect(() => {
    (async () => {
      try {
        const data = await strategyApi.GetList();
        setStrategies(data);
      } catch {
        setStrategies([]);
      }
    })();
  }, []);

  /* ─── Filtered runs ─── */
  const filtered = useMemo(() => {
    return runs.filter((r) => {
      if (search.trim()) {
        const q = search.toLowerCase();
        const match =
          r.name.toLowerCase().includes(q) ||
          r.strategyKey.toLowerCase().includes(q);
        if (!match) return false;
      }
      if (stateFilter !== "" && r.state !== Number(stateFilter)) {
        return false;
      }
      return true;
    });
  }, [runs, search, stateFilter]);

  /* ─── Handlers ─── */
  const openCreate = () => {
    setFormMode("create");
    setFormDetail(null);
    setFormOpen(true);
  };

  const openEdit = async (run: TradingRunInfoView) => {
    try {
      const detail = await tradingRunApi.GetById(run.id);
      setFormDetail(detail);
      setFormMode("edit");
      setFormOpen(true);
    } catch (e) {
      alert(
        "خطا در دریافت جزئیات: " +
          (e instanceof Error ? e.message : "نامشخص"),
      );
    }
  };

  const submitForm = async (
    cmd: CreateTradingRunCommand | UpdateTradingRunCommand,
  ) => {
    setFormSaving(true);
    try {
      if ("id" in cmd) {
        await handleUpdate(cmd);
      } else {
        await handleCreate(cmd);
      }
      setFormOpen(false);
    } finally {
      setFormSaving(false);
    }
  };

  const confirmStop = async (behavior: StopRunBehavior) => {
    if (!stopTarget) return;
    setStopSaving(true);
    try {
      await handleStop(stopTarget.id, behavior);
      setStopTarget(null);
    } finally {
      setStopSaving(false);
    }
  };

  const confirmDelete = async () => {
    if (!deleteTarget) return;
    setDeleteSaving(true);
    try {
      await handleDelete(deleteTarget.id);
      setDeleteTarget(null);
    } finally {
      setDeleteSaving(false);
    }
  };

  /* ─── Render ─── */
  return (
    <div className="space-y-4 p-6">
      {/* Header */}
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-xl font-bold text-gray-900 dark:text-white">
            پلن‌های معاملاتی
          </h1>
          <p className="mt-1 text-xs text-gray-500 dark:text-gray-400">
            {runs.length} پلن · {filtered.length} نمایش داده شده
          </p>
        </div>
        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={refresh}
            disabled={loading}
            className="rounded-lg border border-gray-300 px-3 py-2 text-sm text-gray-700 hover:bg-gray-100 disabled:opacity-50 dark:border-gray-600 dark:text-gray-300 dark:hover:bg-gray-700"
          >
            🔄 بروزرسانی
          </button>
          <button
            type="button"
            onClick={openCreate}
            className="rounded-lg bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700"
          >
            ➕ پلن جدید
          </button>
        </div>
      </div>

      {/* Filters */}
      <div className="flex flex-wrap items-center gap-2">
        <input
          type="text"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          placeholder="جستجو (نام یا استراتژی)..."
          className="flex-1 min-w-[200px] rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
        <select
          value={stateFilter}
          onChange={(e) => setStateFilter(e.target.value)}
          className="rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        >
          <option value="">همه‌ی وضعیت‌ها</option>
          {Object.entries(TRADING_RUN_STATE_LABELS).map(([k, v]) => (
            <option key={k} value={k}>
              {v}
            </option>
          ))}
        </select>
      </div>

      {/* Error */}
      {error && (
        <div className="rounded-lg bg-red-50 p-3 text-sm text-red-700 dark:bg-red-900/30 dark:text-red-400">
          ⚠️ {error}
        </div>
      )}

      {/* Table */}
      <div className="overflow-x-auto rounded-lg border border-gray-200 shadow-sm dark:border-gray-700">
        <table className="w-full text-right text-sm text-gray-700 dark:text-gray-300">
          <thead className="bg-gray-50 text-xs text-gray-700 dark:bg-gray-700 dark:text-gray-300">
            <tr>

  onStop: () => void;

  onDelete: () => void;
}

  onStart,
  onStop,
  onEdit,
  onDelete,
}) => {
  const state = run.state as TradingRunState;

    state === TradingRunState.Ready || state === TradingRunState.Stopped;
  const canStop = state === TradingRunState.Running;

  const canDelete = state !== TradingRunState.Running;

  const btn =
    "rounded px-2.5 py-1 text-xs font-medium disabled:opacity-40 whitespace-nowrap";

  return (
    <div className="flex flex-wrap justify-end gap-1">

          title="شروع"

          type="button"

          title="حذف"
        >
          🗑 حذف
        </button>
      )}
    </div>
  );
};

import React from "react";
import type { StrategyConfigFieldView } from "../../../models";

interface Props {
  fields: StrategyConfigFieldView[];
  values: Record<string, any>;
  onChange: (name: string, value: any) => void;
}

type InputKind = "number" | "text" | "boolean";

function getInputKind(typeName: string): InputKind {
  const t = typeName.toLowerCase();
  if (
    t.includes("int") ||
    t.includes("decimal") ||
    t.includes("double") ||
    t.includes("float") ||
    t.includes("long")
  ) return "number";
  if (t === "boolean" || t === "bool") return "boolean";
  return "text";
}

function isDecimal(typeName: string): boolean {
  const t = typeName.toLowerCase();
  return t.includes("decimal") || t.includes("double") || t.includes("float");
}

export const StrategyConfigEditor: React.FC<Props> = ({
  fields,
  values,
  onChange,
}) => {
  if (fields.length === 0) {
    return (
      <p className="rounded bg-slate-100 p-3 text-xs text-slate-600 dark:bg-slate-700 dark:text-slate-300">
        این استراتژی پارامتر قابل تنظیمی ندارد
      </p>
    );
  }

  return (
    <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
      {fields.map((f) => {
        const kind = getInputKind(f.typeName);
        const raw = values[f.name];
        const current = raw ?? f.defaultValue ?? "";

        return (
          <div key={f.name} className="flex flex-col gap-1">
            <label className="text-xs font-medium text-gray-700 dark:text-gray-300">
              <span className="font-mono">{f.name}</span>
              {f.isRequired && <span className="mr-1 text-red-500">*</span>}
              <span className="mr-1 text-[10px] font-normal text-gray-500">
                ({f.typeName})
              </span>
            </label>

            {kind === "boolean" ? (
              <input
                type="checkbox"
                checked={!!current}
                onChange={(e) => onChange(f.name, e.target.checked)}
                className="h-5 w-5 rounded border-gray-300 text-blue-600"
              />
            ) : (
              <input
                type={kind === "number" ? "number" : "text"}
                value={String(current)}
                step={
                  kind === "number" && isDecimal(f.typeName) ? "0.01" : "1"
                }
                onChange={(e) => {
                  const v = e.target.value;
                  onChange(f.name, kind === "number" ? Number(v) : v);
                }}
                dir="ltr"
                className="w-full rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
              />
            )}
          </div>
        );
      })}
    </div>
  );
};

import React, { useEffect, useState } from "react";
import { instrumentApi } from "../../../api/instrumentApi";
import type { TradingRunSymbolModel } from "../../../models";
import type { SelectionListDto } from "@/core/models/SelectionListDto";

interface Props {
  symbols: TradingRunSymbolModel[];
  onChange: (symbols: TradingRunSymbolModel[]) => void;
}

export const SymbolAllocationEditor: React.FC<Props> = ({
  symbols,
  onChange,
}) => {
  const [options, setOptions] = useState<SelectionListDto[]>([]);
  const [selectedIsin, setSelectedIsin] = useState<string>("");
  const [percentage, setPercentage] = useState<string>("");
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    (async () => {
      try {
        const list = await instrumentApi.GetSelectionList();
        setOptions(list);
      } catch {
        setOptions([]);
      } finally {
        setLoading(false);
      }
    })();
  }, []);

  const add = () => {
    if (!selectedIsin) return;
    const pct = Number(percentage);
    if (isNaN(pct) || pct <= 0) {
      alert("درصد باید بزرگ‌تر از صفر باشد");
      return;
    }
    if (symbols.some((s) => s.isin === selectedIsin)) {
      alert("این نماد قبلاً اضافه شده است");
      return;
    }
    onChange([...symbols, { isin: selectedIsin, allocationPercentage: pct }]);
    setSelectedIsin("");
    setPercentage("");
  };

  const remove = (isin: string) => {
    onChange(symbols.filter((s) => s.isin !== isin));
  };

  const update = (isin: string, pct: number) => {
    onChange(
      symbols.map((s) =>
        s.isin === isin ? { ...s, allocationPercentage: pct } : s,
      ),
    );
  };

  const labelFor = (isin: string): string => {
    const opt = options.find((o) => o.value === isin);
    return opt ? opt.display || opt.label : isin;
  };

  const total = symbols.reduce((sum, s) => sum + s.allocationPercentage, 0);
  const totalOk = Math.abs(total - 100) < 0.01;

  const availableOptions = options.filter(
    (o) => !symbols.some((s) => s.isin === o.value),
  );

  return (
    <div className="space-y-3">
      {/* ─── افزودن ─── */}
      <div className="flex flex-col gap-2 sm:flex-row">
        <select
          value={selectedIsin}
          onChange={(e) => setSelectedIsin(e.target.value)}
          disabled={loading}
          className="flex-1 rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        >
          <option value="">
            {loading ? "در حال بارگذاری..." : "انتخاب نماد..."}
          </option>
          {availableOptions.map((o) => (
            <option key={o.value} value={o.value}>
              {o.display || o.label}
            </option>
          ))}
        </select>
        <input
          type="number"
          placeholder="درصد"
          value={percentage}
          onChange={(e) => setPercentage(e.target.value)}
          min={0}
          max={100}
          step="0.01"
          dir="ltr"
          className="w-full rounded border border-gray-300 p-2 text-sm sm:w-24 dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
        <button
          type="button"
          onClick={add}
          disabled={!selectedIsin || !percentage}
          className="rounded bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700 disabled:opacity-50"
        >
          افزودن
        </button>
      </div>

      {/* ─── لیست ─── */}
      {symbols.length === 0 ? (
        <p className="rounded border border-dashed border-gray-300 p-4 text-center text-xs text-gray-500 dark:border-gray-600">
          هیچ نمادی اضافه نشده است
        </p>
      ) : (
        <div className="overflow-hidden rounded border border-gray-200 dark:border-gray-700">
          <div className="flex items-center justify-between bg-slate-100 px-3 py-2 text-xs font-medium dark:bg-slate-700">
            <span>جمع تخصیص:</span>
            <span className={totalOk ? "text-emerald-600" : "text-red-600"}>
              {total.toFixed(2)}%{" "}
              {totalOk ? "✅" : "⚠️ باید ۱۰۰٪ باشد"}
            </span>
          </div>
          <div className="divide-y divide-gray-200 dark:divide-gray-700">
            {symbols.map((s) => (
              <div key={s.isin} className="flex items-center gap-2 p-2">
                <span className="flex-1 text-sm">
                  {labelFor(s.isin)}
                  <span className="mr-1 font-mono text-[10px] text-gray-400">
                    ({s.isin})
                  </span>
                </span>
                <input
                  type="number"
                  value={s.allocationPercentage}
                  onChange={(e) => update(s.isin, Number(e.target.value))}
                  min={0}
                  max={100}
                  step="0.01"
                  dir="ltr"
                  className="w-24 rounded border border-gray-300 p-1 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
                />
                <span className="text-xs text-gray-500">%</span>
                <button
                  type="button"
                  onClick={() => remove(s.isin)}
                  className="rounded p-1 text-xs text-red-600 hover:bg-red-50 dark:hover:bg-red-900/30"
                >
                  حذف
                </button>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
};

export const TradingRunFormModal: React.FC<Props> = ({
  isOpen,
  mode,
  initialDetail,
  strategies,
  onClose,
  onSubmit,
  saving,
}) => {
  /* ─── State ─── */
  const [name, setName] = useState("");
  const [strategyKey, setStrategyKey] = useState("");
  const [strategyConfig, setStrategyConfig] = useState<Record<string, any>>({});
  const [primaryTimeFrame, setPrimaryTimeFrame] = useState<number>(
    TimeFrame.D1,
  );
const [targetCurrency, setTargetCurrency] = useState<string>("");
const [strengthMultiplier, setStrengthMultiplier] = useState<string>("1.5");
const [minDeltaPercent, setMinDeltaPercent] = useState<string>("3");  
const [executionMode, setExecutionMode] = useState<number>(ExecutionMode.Live); 
const [currencyRateStatus, setCurrencyRateStatus] =
  useState<"idle" | "checking" | "ok" | "missing">("idle");
const [currencyRateValue, setCurrencyRateValue] = useState<number | null>(null);
  const [capitalAllocationMode, setCapitalAllocationMode] = useState<number>(
    CapitalAllocationMode.Fixed,
  );
  const [fixedCapitalAmount, setFixedCapitalAmount] = useState<string>("");

  const [slMode, setSlMode] = useState<SlTpMode>("none");
  const [slPercent, setSlPercent] = useState<string>("");
  const [slPrice, setSlPrice] = useState<string>("");

  const [tpMode, setTpMode] = useState<SlTpMode>("none");
  const [tpPercent, setTpPercent] = useState<string>("");
  const [tpPrice, setTpPrice] = useState<string>("");

  const [defaultOrderMode, setDefaultOrderMode] = useState<number>(
    OrderMode.Quantity,
  );
  const [onMarketClose, setOnMarketClose] = useState<number>(
    MarketCloseBehavior.KeepPositions,
  );
  const [onStop, setOnStop] = useState<number>(StopRunBehavior.KeepPositions);

  const [symbols, setSymbols] = useState<TradingRunSymbolModel[]>([]);
  const [accountIds, setAccountIds] = useState<string[]>([]);

  const [error, setError] = useState<string | null>(null);
useEffect(() => {
  if (!isOpen) return;

  const cur = targetCurrency.trim().toUpperCase();
  if (!cur || cur === "IRR") {
    setCurrencyRateStatus("idle");
    setCurrencyRateValue(null);
    return;
  }

  let cancelled = false;
  setCurrencyRateStatus("checking");
  setCurrencyRateValue(null);

  (async () => {
    try {
      const rate = await currencyRateApi.GetLatest(cur, "IRR");
      if (cancelled) return;
      if (rate != null && rate > 0) {
        setCurrencyRateStatus("ok");
        setCurrencyRateValue(rate);
      } else {
        // Try reverse direction (IRR→cur) in case only that exists
        const reverseRate = await currencyRateApi.GetLatest("IRR", cur);
        if (cancelled) return;
        if (reverseRate != null && reverseRate > 0) {
          setCurrencyRateStatus("ok");
          setCurrencyRateValue(1 / reverseRate);
        } else {
          setCurrencyRateStatus("missing");
        }
      }
    } catch {
      if (!cancelled) setCurrencyRateStatus("missing");
    }
  })();

  return () => {
    cancelled = true;
  };
}, [targetCurrency, isOpen]);

  /* ─── Init: Edit ─── */
  useEffect(() => {
    if (!isOpen) return;

    if (mode === "edit" && initialDetail) {
      setName(initialDetail.name);
      setStrategyKey(initialDetail.strategyKey);
      setPrimaryTimeFrame(initialDetail.primaryTimeFrame);
      setTargetCurrency(initialDetail.targetCurrency ?? "");
      setStrengthMultiplier(initialDetail.strengthMultiplier.toString());
setMinDeltaPercent((initialDetail.minDeltaPercent * 100).toString()); 
setExecutionMode(initialDetail.executionMode);
      setCapitalAllocationMode(initialDetail.settings.capitalAllocationMode);
      setFixedCapitalAmount(
        initialDetail.settings.fixedCapitalAmount?.toString() ?? "",
      );

      if (initialDetail.settings.stopLossPercentage != null) {

        setSlPercent(

        setSlPrice("");

        setSlMode("price");
        setSlPercent("");
        setSlPrice(initialDetail.settings.stopLossPrice.toString());
      } else {
        setSlMode("none");
        setSlPercent("");
        setSlPrice("");
      }

      if (initialDetail.settings.takeProfitPercentage != null) {
        setTpMode("percent");
        setTpPercent(
          initialDetail.settings.takeProfitPercentage.toString(),
        );
        setTpPrice("");
      } else if (initialDetail.settings.takeProfitPrice != null) {
        setTpMode("price");
        setTpPercent("");
        setTpPrice(initialDetail.settings.takeProfitPrice.toString());
      } else {
        setTpMode("none");
        setTpPercent("");
        setTpPrice("");
      }

      setDefaultOrderMode(initialDetail.settings.defaultOrderMode);
      setOnMarketClose(initialDetail.settings.onMarketClose);
      setOnStop(initialDetail.settings.onStop);
      setSymbols(initialDetail.symbols);
      setAccountIds(initialDetail.accountIds);
      setError(null);

      try {
        setStrategyConfig(JSON.parse(initialDetail.strategyConfigJson));
      } catch {
        setStrategyConfig({});
      }
      return;
    }

    /* ─── Create ─── */
    setName("");
    setStrategyKey(strategies[0]?.key ?? "");
    setPrimaryTimeFrame(TimeFrame.D1);
    setTargetCurrency("");
    setStrengthMultiplier("1.5");
setMinDeltaPercent("3");   
setExecutionMode(ExecutionMode.Live);
setCurrencyRateStatus("idle");
setCurrencyRateValue(null);
    setCapitalAllocationMode(CapitalAllocationMode.Fixed);
    setFixedCapitalAmount("");
    setSlMode("none");
    setSlPercent("");
    setSlPrice("");
    setTpMode("none");
    setTpPercent("");
    setTpPrice("");
    setDefaultOrderMode(OrderMode.Quantity);
    setOnMarketClose(MarketCloseBehavior.KeepPositions);
    setOnStop(StopRunBehavior.KeepPositions);
    setSymbols([]);
    setAccountIds([]);
    setError(null);
  }, [isOpen, mode, initialDetail, strategies]);

  /* ─── Strategy config: init from defaults when strategy changes (Create) ─── */
  useEffect(() => {
    if (!isOpen || mode !== "create") return;
    const strategy = strategies.find((s) => s.key === strategyKey);
    if (!strategy) {

    }
    const init: Record<string, any> = {};
    strategy.configFields.forEach((f) => {

      const t = f.typeName.toLowerCase();
      if (
        t.includes("int") ||
        t.includes("decimal") ||
        t.includes("double") ||
        t.includes("float") ||
        t.includes("long")

        init[f.name] = Number(f.defaultValue);
      } else if (t === "boolean" || t === "bool") {
        init[f.name] = f.defaultValue === "true";
      } else {
        init[f.name] = f.defaultValue;
      }
    });
    setStrategyConfig(init);

  if (!isOpen) return null;

  const selectedStrategy = strategies.find((s) => s.key === strategyKey);

  /* ─── Validation ─── */
  const validate = (): string | null => {
    if (!name.trim()) return "نام پلن الزامی است";

    const total = symbols.reduce((s, x) => s + x.allocationPercentage, 0);

    if (accountIds.length === 0)

  executionMode,                                          // ← NEW

  symbols,
  accountIds,

          <h3 className="text-lg font-bold text-gray-900 dark:text-white">
            {mode === "edit" ? "ویرایش پلن معاملاتی" : "ایجاد پلن معاملاتی جدید"}
          </h3>
          <button
            type="button"
            onClick={onClose}

          <Section title="اطلاعات پایه">
            <div className="flex flex-col gap-1">
              <label className="text-xs font-medium text-gray-700 dark:text-gray-300">
                نام پلن <span className="text-red-500">*</span>
              </label>
              <input
                type="text"
                value={name}
                onChange={(e) => setName(e.target.value)}
                placeholder="مثلاً: RSI Daily فولاد و شپنا"
                className="w-full rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
              />
            </div>
          </Section>
{/* ۱.۵ Execution Mode */}
<Section title="حالت اجرا">
  <div className="grid grid-cols-1 gap-2 md:grid-cols-2">
    {[ExecutionMode.Live, ExecutionMode.Paper].map((mode) => {
      const isSelected = executionMode === mode;
      const isLive = mode === ExecutionMode.Live;

      return (
        <label
          key={mode}
          className={`flex cursor-pointer items-start gap-3 rounded-lg border p-3 transition-colors ${
            isSelected
              ? isLive
                ? "border-red-500 bg-red-50 dark:bg-red-900/20"
                : "border-cyan-500 bg-cyan-50 dark:bg-cyan-900/20"
              : "border-gray-200 hover:bg-gray-50 dark:border-gray-700 dark:hover:bg-gray-700/30"
          }`}
        >
          <input
            type="radio"
            name="executionMode"
            checked={isSelected}
            onChange={() => setExecutionMode(mode)}
            className="mt-1"
          />
          <div>
            <p className="text-sm font-medium text-gray-900 dark:text-white">
              {EXECUTION_MODE_LABELS[mode]}
            </p>
            <p className="mt-0.5 text-[10px] text-gray-500 dark:text-gray-400">
              {isLive
                ? "سفارش واقعی به کارگزاری ارسال می‌شود. سرمایه‌ی واقعی در ریسک است."
                : "سفارش فقط شبیه‌سازی می‌شود. بدون ارتباط با کارگزاری، بدون ریسک پول واقعی."}
            </p>
          </div>
        </label>
      );
    })}
  </div>

  {executionMode === ExecutionMode.Paper && (
    <div className="mt-2 rounded-lg border border-cyan-700 bg-cyan-900/20 p-3 text-xs text-cyan-300">
      ℹ️ در حالت Paper، سفارش‌ها بلافاصله در قیمت Best Ask/Bid پر می‌شوند.
      Commission و Slippage روی حساب مجازی محاسبه می‌شوند ولی پول واقعی خارج
      نمی‌شود. تمام Orderها/Positionها با تگ <strong>Paper</strong> در DB ثبت
      می‌شوند.
    </div>
  )}
</Section>
          {/* ۲. Strategy */}
          <Section title="استراتژی">
            <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
              {/* Currency */}
<div className="mt-3">
  <label className="mb-1 block text-xs font-medium text-gray-700 dark:text-gray-300">
    ارز تحلیل (اختیاری)
  </label>
  <div className="flex items-center gap-2">
    <select
      value={targetCurrency || ""}
      onChange={(e) => setTargetCurrency(e.target.value)}
      className="w-full rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
    >
      <option value="">— بدون تبدیل (ریال) —</option>
      <option value="USD">دلار آمریکا (USD)</option>
      <option value="EUR">یورو (EUR)</option>
      <option value="GBP">پوند انگلیس (GBP)</option>
      <option value="AED">درهم امارات (AED)</option>
      <option value="TRY">لیر ترکیه (TRY)</option>
      <option value="CNY">یوان چین (CNY)</option>
      <option value="JPY">ین ژاپن (JPY)</option>
      <option value="CHF">فرانک سوئیس (CHF)</option>
      <option value="GOLD">طلا (GOLD)</option>
    </select>
    {currencyRateStatus === "checking" && (
      <span className="whitespace-nowrap text-xs text-gray-500">
        ⏳ بررسی...
      </span>
    )}
    {currencyRateStatus === "ok" && currencyRateValue != null && (
      <span className="whitespace-nowrap text-xs text-emerald-500">
        ✓ {currencyRateValue.toLocaleString("en-US", { maximumFractionDigits: 2 })}
      </span>
    )}
    {currencyRateStatus === "missing" && (
      <span className="whitespace-nowrap text-xs text-red-500">
        ❌ نرخ ثبت نشده
      </span>
    )}
  </div>
  <p className="mt-1 text-[10px] text-gray-500">
    اگر خالی باشد، تحلیل روی قیمت ریالی انجام می‌شود. 
    برای ارز دیگر، ابتدا از صفحه‌ی «نرخ‌های ارز» نرخ ثبت کنید.
  </p>
</div>
{/* Strength Multiplier */}
<div className="mt-3">
  <label className="mb-1 block text-xs font-medium text-gray-700 dark:text-gray-300">
    ضریب قدرت سیگنال (Stepped Buying)
  </label>
  <div className="flex items-center gap-2">
    <input
      type="number"
      value={strengthMultiplier}
      onChange={(e) => setStrengthMultiplier(e.target.value)}
      dir="ltr"
      min={0.1}
      max={5}
      step={0.1}
      className="w-32 rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"

/* ═══════════════════════════════════════════════════════
   دکمه‌ی لاگین همه
   ═══════════════════════════════════════════════════════ */
interface LoginAllResult {
  accountId: string;
  accountName: string;
  ok: boolean;
  error?: string;
}

const LoginAllButton: React.FC<{
  accounts: AccountInfoView[];
  onComplete: () => void;
}> = ({ accounts, onComplete }) => {
  const [busy, setBusy] = useState(false);
  const [results, setResults] = useState<LoginAllResult[] | null>(null);
  const [progress, setProgress] = useState<{ current: number; total: number } | null>(null);

  const handleLoginAll = async () => {
    if (accounts.length === 0) {
      alert("هیچ حسابی وجود ندارد");
      return;
    }

    if (!confirm(`لاگین همه‌ی ${accounts.length} حساب؟`)) return;

    setBusy(true);
    setResults(null);
    setProgress({ current: 0, total: accounts.length });

    const list: LoginAllResult[] = [];

    /* ─── لوپ سریال — چون هر login سنگین است (OIDC flow) ─── */
    for (let i = 0; i < accounts.length; i++) {
      const acc = accounts[i];
      setProgress({ current: i + 1, total: accounts.length });

      try {
        await accountApi.login(acc.id);
        list.push({
          accountId: acc.id,
          accountName: acc.name,
          ok: true,
        });
      } catch (e) {
        list.push({
          accountId: acc.id,
          accountName: acc.name,
          ok: false,
          error: e instanceof Error ? e.message : "خطای نامشخص",
        });
      }
    }

    setResults(list);
    setBusy(false);
    setProgress(null);
    onComplete();
  };

  const successCount = results?.filter((r) => r.ok).length ?? 0;
  const failCount = results?.filter((r) => !r.ok).length ?? 0;

  return (
    <div className="mb-4 rounded-xl border border-slate-700 bg-slate-900 p-4">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex items-center gap-3">
          <button
            type="button"
            onClick={handleLoginAll}
            disabled={busy || accounts.length === 0}
            className="rounded-lg bg-emerald-600 px-4 py-2 text-sm font-semibold text-white hover:bg-emerald-500 disabled:opacity-50"
          >
            {busy
              ? `⏳ در حال لاگین... (${progress?.current}/${progress?.total})`
              : "🔐 لاگین همه"}
          </button>

          <span className="text-xs text-slate-400">
            {accounts.length} حساب
          </span>
        </div>

        {results && (
          <div className="flex flex-wrap gap-2 text-xs">
            <span className="rounded bg-emerald-900/40 px-2 py-1 text-emerald-300">
              ✅ {successCount} موفق
            </span>
            {failCount > 0 && (
              <span className="rounded bg-red-900/40 px-2 py-1 text-red-300">
                ❌ {failCount} ناموفق
              </span>
            )}
          </div>
        )}
      </div>

      {/* ─── گزارش نتیجه ─── */}
      {results && results.length > 0 && (
        <div className="mt-3 space-y-1 border-t border-slate-700 pt-3">
          {results.map((r) => (
            <div
              key={r.accountId}
              className={`flex items-center justify-between rounded px-2 py-1 text-xs ${
                r.ok
                  ? "bg-emerald-950/20 text-emerald-300"
                  : "bg-red-950/20 text-red-300"
              }`}
            >
              <span className="font-medium">{r.accountName}</span>
              <span className="font-mono text-[11px]">
                {r.ok ? "✅ موفق" : `❌ ${r.error}`}
              </span>
            </div>
          ))}
        </div>
      )}
    </div>
  );
};

/* ═══════════════════════════════════════════════════════
   صفحه‌ی اصلی
   ═══════════════════════════════════════════════════════ */
export const AccountsManagementPage: React.FC = () => {
  /* برای گرفتن لیست حساب‌ها قبل از رندر دکمه */
  const [accounts, setAccounts] = useState<AccountInfoView[]>([]);
  const [reloadKey, setReloadKey] = useState(0);

  const loadAccounts = React.useCallback(async () => {
    try {
      const list = await accountApi.GetList();
      setAccounts(list);
    } catch {
      setAccounts([]);
    }
  }, []);

  React.useEffect(() => {
    void loadAccounts();
  }, [loadAccounts, reloadKey]);

  const handleLoginComplete = () => {
    /* بعد از لاگین همه، لیست رو refresh کن (sessionStatus عوض شده) */
    void loadAccounts();
    /* و GenericCrudPage رو هم مجبور به refresh کن */
    setReloadKey((k) => k + 1);
  };

  return (
    <div>
      <LoginAllButton accounts={accounts} onComplete={handleLoginComplete} />

      <GenericCrudPage<
        AccountInfoView,
        CreateAccountCommand,
        UpdateAccountCommand
      >
        key={reloadKey}
        title="مدیریت حساب‌ها"
        columns={columns}
        crudOptions={{
          api: accountCrudApi,
          columns,
          mapToUpdateCommand: (item) => ({
            id: item.id,
            name: item.name,
            username: item.username,
            password: undefined,
          }),
          mapToCreateCommand: (formData) => ({
            broker: 1,
            name: formData.name,
            username: formData.username,
            password: formData.password ?? "",
          }),
          tableFeatures: {
            enableSearch: true,
            enableDelete: true,
            enableColumnFilter: false,
          },
          pageFeatures: {
            enableAdd: true,
          },
        }}
      />
    </div>
  );
};

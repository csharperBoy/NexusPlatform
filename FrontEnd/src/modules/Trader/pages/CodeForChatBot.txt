//src/modules/Trader/pages/Backtests/components/BacktestLogsTab.tsx
import React, { useEffect, useMemo, useState } from "react";
import { backtestApi } from "../../../api/backtestApi";
import type { BacktestEventView, BacktestEventQuery } from "../../../models";
import {
  BACKTEST_EVENT_TYPE_LABELS,
  BACKTEST_EVENT_TYPE_COLORS,
  BACKTEST_EVENT_SEVERITY_LABELS,
  BacktestEventType,
} from "../../../models";

interface Props {
  backtestRunId: string;
}

const parseUtcDate = (s: string): Date => {
  const hasTz = /[Zz]|[+-]\d{2}:?\d{2}$/.test(s);
  return new Date(hasTz ? s : s + "Z");
};

const fmtDateTime = (s: string): string =>
  parseUtcDate(s).toLocaleString("fa-IR", {
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
  });

const prettyJson = (s: string | null): string => {
  if (!s) return "";
  try {
    return JSON.stringify(JSON.parse(s), null, 2);
  } catch {
    return s;
  }
};

export const BacktestLogsTab: React.FC<Props> = ({ backtestRunId }) => {
  const [events, setEvents] = useState<BacktestEventView[]>([]);
  const [totalRecords, setTotalRecords] = useState(0);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const [typeFilter, setTypeFilter] = useState<string>("");
  const [severityFilter, setSeverityFilter] = useState<string>("");
  const [search, setSearch] = useState<string>("");

  const [selectedEvent, setSelectedEvent] = useState<BacktestEventView | null>(
    null,
  );

  const query: BacktestEventQuery = useMemo(
    () => ({
      backtestRunId,
      type: typeFilter !== "" ? Number(typeFilter) : null,
      minSeverity: severityFilter !== "" ? Number(severityFilter) : null,
      search: search.trim() || null,
      skip: 0,
      take: 5000,
    }),
    [backtestRunId, typeFilter, severityFilter, search],
  );

  useEffect(() => {
    (async () => {
      setLoading(true);
      setError(null);
      try {
        const result = await backtestApi.QueryEvents(query);
        setEvents(result.items);
        setTotalRecords(result.totalRecords);
      } catch (e) {
        setError(e instanceof Error ? e.message : "خطا در دریافت رویدادها");
        setEvents([]);
      } finally {
        setLoading(false);
      }
    })();
  }, [query]);

  const typeOptions = useMemo(() => {
    const present = new Set(events.map((e) => e.type));
    return Array.from(present).sort((a, b) => a - b);
  }, [events]);

  return (
    <div className="space-y-3">
      {/* Filters */}
      <div className="grid grid-cols-1 gap-2 md:grid-cols-3">
        <select
          value={typeFilter}
          onChange={(e) => setTypeFilter(e.target.value)}
          className="rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        >
          <option value="">همه‌ی نوع‌ها</option>
          {typeOptions.map((t) => (
            <option key={t} value={t}>
              {BACKTEST_EVENT_TYPE_LABELS[t] ?? t}
            </option>
          ))}
        </select>

        <select
          value={severityFilter}
          onChange={(e) => setSeverityFilter(e.target.value)}
          className="rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        >
          <option value="">همه‌ی سطوح</option>
          <option value="0">Debug و بالاتر</option>
          <option value="1">Info و بالاتر</option>
          <option value="2">Warning و بالاتر</option>
          <option value="3">فقط Error و Critical</option>
        </select>

        <input
          type="text"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          placeholder="جستجو در پیام..."
          className="rounded border border-gray-300 p-2 text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
      </div>

      <p className="text-xs text-gray-500 dark:text-gray-400">
        {totalRecords.toLocaleString("fa-IR")} رویداد
      </p>

      {error && (
        <div className="rounded-lg bg-red-50 p-3 text-sm text-red-700 dark:bg-red-900/30 dark:text-red-400">
          ⚠️ {error}
        </div>
      )}

      {/* Table */}
      <div className="max-h-[600px] overflow-y-auto overflow-x-auto rounded-lg border border-gray-200 dark:border-gray-700">
        <table className="w-full text-right text-xs text-gray-700 dark:text-gray-300">
          <thead className="sticky top-0 z-10 bg-gray-100 text-xs text-gray-700 dark:bg-gray-700 dark:text-gray-300">
            <tr>
              <th className="p-2">زمان</th>
              <th className="p-2">نوع</th>
              <th className="p-2">سطح</th>
              <th className="p-2">پیام</th>
              <th className="p-2">ISIN</th>
              <th className="p-2">کندل</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {loading ? (
              <tr>
                <td colSpan={6} className="p-6 text-center">
                  در حال بارگذاری...
                </td>
              </tr>
            ) : events.length === 0 ? (
              <tr>
                <td colSpan={6} className="p-6 text-center text-gray-500">
                  رویدادی یافت نشد.
                </td>
              </tr>
            ) : (
              events.map((e) => (
                <tr
                  key={e.id}
                  onClick={() => setSelectedEvent(e)}
                  className="cursor-pointer hover:bg-gray-50 dark:hover:bg-gray-700/30"
                >
                  <td className="whitespace-nowrap p-2 font-mono text-[10px]">
                    {fmtDateTime(e.occurredAt)}
                  </td>
                  <td className="p-2">
                    <span
                      className={`inline-flex rounded px-1.5 py-0.5 text-[10px] font-medium whitespace-nowrap ${
                        BACKTEST_EVENT_TYPE_COLORS[e.type] ?? ""
                      }`}
                    >
                      {BACKTEST_EVENT_TYPE_LABELS[e.type] ?? e.typeName}
                    </span>
                  </td>
                  <td className="p-2 text-[10px]">
                    {BACKTEST_EVENT_SEVERITY_LABELS[e.severity] ?? e.severityName}
                  </td>
                  <td className="p-2">{e.message}</td>
                  <td className="p-2 font-mono text-[10px]">{e.isin ?? "—"}</td>
                  <td className="whitespace-nowrap p-2 text-[10px] text-gray-500">
                    {e.candleTime ? fmtDateTime(e.candleTime) : "—"}
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>

      {/* Detail Drawer */}
      {selectedEvent && (
        <div
          className="fixed inset-0 z-50 flex justify-end bg-black/50"
          onClick={() => setSelectedEvent(null)}
        >
          <div
            className="w-full max-w-2xl overflow-y-auto bg-white shadow-xl dark:bg-gray-800"
            onClick={(e) => e.stopPropagation()}
          >
            <div className="sticky top-0 flex items-center justify-between border-b border-gray-200 bg-white px-5 py-3 dark:border-gray-700 dark:bg-gray-800">
              <h3 className="text-lg font-bold text-gray-900 dark:text-white">
                جزئیات رویداد
              </h3>
              <button
                onClick={() => setSelectedEvent(null)}
                className="text-2xl text-gray-500 hover:text-gray-700"
              >
                ×
              </button>
            </div>
            <div className="space-y-4 p-5">
              <div className="grid grid-cols-2 gap-3 text-sm">
                <div>
                  <p className="text-xs text-gray-500">زمان</p>
                  <p className="font-mono text-xs">
                    {fmtDateTime(selectedEvent.occurredAt)}
                  </p>
                </div>
                <div>
                  <p className="text-xs text-gray-500">نوع</p>
                  <p>{BACKTEST_EVENT_TYPE_LABELS[selectedEvent.type] ?? selectedEvent.typeName}</p>
                </div>
                <div>
                  <p className="text-xs text-gray-500">سطح</p>
                  <p>{BACKTEST_EVENT_SEVERITY_LABELS[selectedEvent.severity] ?? selectedEvent.severityName}</p>
                </div>
                {selectedEvent.isin && (
                  <div>
                    <p className="text-xs text-gray-500">ISIN</p>
                    <p className="font-mono text-xs">{selectedEvent.isin}</p>
                  </div>
                )}
                {selectedEvent.candleTime && (
                  <div>
                    <p className="text-xs text-gray-500">زمان کندل</p>
                    <p className="font-mono text-xs">
                      {fmtDateTime(selectedEvent.candleTime)}
                    </p>
                  </div>
                )}
              </div>

              <div className="rounded-lg bg-gray-50 p-3 dark:bg-gray-900/40">
                <p className="text-xs text-gray-500">پیام</p>
                <p className="mt-1 text-sm text-gray-900 dark:text-gray-100">
                  {selectedEvent.message}
                </p>
              </div>

              {selectedEvent.payloadJson && (
                <div>
                  <p className="mb-1 text-xs font-semibold text-gray-700 dark:text-gray-300">
                    Payload (JSON)
                  </p>
                  <pre
                    dir="ltr"
                    className="max-h-96 overflow-auto rounded-lg bg-gray-900 p-3 text-xs text-gray-100"
                  >
                    {prettyJson(selectedEvent.payloadJson)}
                  </pre>
                </div>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

//src/modules/Trader/pages/Backtests/components/EquityCurveChart.tsx
import React, { useMemo } from "react";
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
  ResponsiveContainer,
} from "recharts";
import type {
  NameType,
  ValueType,
} from "recharts/types/component/DefaultTooltipContent";
import type { BacktestDailyEquityView } from "../../../models";

interface Series {
  name: string;
  data: BacktestDailyEquityView[];
  color: string;
}

interface Props {
  series: Series[];
}

const COLORS = [
  "#10b981",
  "#3b82f6",
  "#f59e0b",
  "#ef4444",
  "#8b5cf6",
];

export const EquityCurveChart: React.FC<Props> = ({ series }) => {
  const mergedData = useMemo(() => {
    const dateMap = new Map<string, Record<string, number | string>>();

    series.forEach((s, idx) => {
      s.data.forEach((point) => {
        const key = point.date.slice(0, 10);
        if (!dateMap.has(key)) {
          dateMap.set(key, { date: key });
        }
        dateMap.get(key)![`series${idx}`] = point.equity;
      });
    });

    return Array.from(dateMap.values()).sort((a, b) =>
      String(a.date).localeCompare(String(b.date)),
    );
  }, [series]);

  if (mergedData.length === 0) {
    return (
      <div className="rounded-lg border border-gray-200 bg-white p-8 text-center text-sm text-gray-500 dark:border-gray-700 dark:bg-gray-800">
        داده‌ای برای نمودار وجود ندارد.
      </div>
    );
  }

  return (
    <div className="rounded-lg border border-gray-200 bg-white p-4 dark:border-gray-700 dark:bg-gray-800">
      <h3 className="mb-4 text-sm font-bold text-gray-900 dark:text-white">
        منحنی سرمایه (Equity Curve)
      </h3>
      <div className="h-80 w-full" dir="ltr">
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={mergedData}>
            <CartesianGrid strokeDasharray="3 3" opacity={0.2} />
            <XAxis
              dataKey="date"
              tick={{ fontSize: 10 }}
              tickFormatter={(v: string | number) => {
                const d = new Date(String(v));
                return `${d.getMonth() + 1}/${d.getDate()}`;
              }}
            />
            <YAxis
              tick={{ fontSize: 10 }}
              tickFormatter={(v: number) => {
                if (Math.abs(v) >= 1_000_000) return `${(v / 1_000_000).toFixed(1)}M`;
                if (Math.abs(v) >= 1_000) return `${(v / 1_000).toFixed(0)}K`;
                return String(v);
              }}
            />
            <Tooltip
              formatter={(value: ValueType | undefined) =>
                typeof value === "number"
                  ? value.toLocaleString("fa-IR", { maximumFractionDigits: 0 })
                  : String(value ?? "")
              }
              labelFormatter={(label) => `تاریخ: ${String(label)}`}
              contentStyle={{
                direction: "rtl",
                fontFamily: "inherit",
                fontSize: 12,
              }}
            />
            <Legend
              formatter={(value: NameType) => {
                const str = String(value);
                const idx = parseInt(str.replace("series", ""), 10);
                return series[idx]?.name ?? str;
              }}
            />
            {series.map((s, idx) => (
              <Line
                key={idx}
                type="monotone"
                dataKey={`series${idx}`}
                stroke={s.color ?? COLORS[idx % COLORS.length]}
                strokeWidth={2}
                dot={false}
                connectNulls
              />
            ))}
          </LineChart>
        </ResponsiveContainer>
      </div>
    </div>
  );
};

  v > 0

import React, { useState } from "react";
import type { BacktestTradeView } from "../../../models";
import { EXIT_REASON_LABELS, BacktestExitReason } from "../../../models";

interface Props {
  trades: BacktestTradeView[];
}

const fmtMoney = (v: number): string =>
  v.toLocaleString("fa-IR", { maximumFractionDigits: 0 });

const parseUtcDate = (s: string): Date => {
  const hasTz = /[Zz]|[+-]\d{2}:?\d{2}$/.test(s);
  return new Date(hasTz ? s : s + "Z");
};

const fmtDate = (s: string): string =>
  parseUtcDate(s).toLocaleString("fa-IR", {
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
  });

const fmtDuration = (sec: number): string => {
  if (sec < 3600) return `${Math.round(sec / 60)}m`;
  if (sec < 86400) return `${(sec / 3600).toFixed(1)}h`;
  return `${(sec / 86400).toFixed(1)}d`;
};

const ExitBadge: React.FC<{ reason: number }> = ({ reason }) => {
  const r = reason as BacktestExitReason;
  const color =
    r === BacktestExitReason.TakeProfit
      ? "bg-emerald-900/40 text-emerald-300"
      : r === BacktestExitReason.StopLoss
        ? "bg-red-900/40 text-red-300"
        : r === BacktestExitReason.Signal
          ? "bg-blue-900/40 text-blue-300"
          : "bg-slate-700 text-slate-300";

  return (
    <span
      className={`inline-flex rounded-full px-2 py-0.5 text-[10px] font-medium whitespace-nowrap ${color}`}
    >
      {EXIT_REASON_LABELS[r] ?? "—"}
    </span>
  );
};

export const TradesTable: React.FC<Props> = ({ trades }) => {
  const [sortBy, setSortBy] = useState<"entry" | "netPnl">("entry");

  const sorted = [...trades].sort((a, b) => {
    if (sortBy === "netPnl") return b.netPnL - a.netPnL;
    return a.entryTime.localeCompare(b.entryTime);
  });

  if (trades.length === 0) {
    return (
      <div className="rounded-lg border border-gray-200 bg-white p-8 text-center text-sm text-gray-500 dark:border-gray-700 dark:bg-gray-800">
        هیچ تریدی در این بکتست ثبت نشده.
      </div>
    );

  return (
    <div className="rounded-lg border border-gray-200 bg-white dark:border-gray-700 dark:bg-gray-800">
      <div className="flex items-center justify-between border-b border-gray-200 p-3 dark:border-gray-700">
        <h3 className="text-sm font-bold text-gray-900 dark:text-white">
          تریدها ({trades.length})
        </h3>
        <div className="flex gap-1">
          <button
            type="button"
            onClick={() => setSortBy("entry")}
            className={`rounded px-2 py-1 text-xs ${
              sortBy === "entry"
                ? "bg-blue-600 text-white"
                : "text-gray-600 dark:text-gray-400"
            }`}
          >
            زمانی
          </button>
          <button
            type="button"
            onClick={() => setSortBy("netPnl")}
            className={`rounded px-2 py-1 text-xs ${
              sortBy === "netPnl"
                ? "bg-blue-600 text-white"
                : "text-gray-600 dark:text-gray-400"
            }`}
          >
            سود
          </button>
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="w-full text-right text-xs">
          <thead className="bg-gray-50 text-gray-700 dark:bg-gray-700 dark:text-gray-300">
            <tr>
              <th className="p-2">نماد</th>
              <th className="p-2">ورود</th>
              <th className="p-2">خروج</th>
              <th className="p-2 text-center">تعداد</th>
              <th className="p-2">میانگین ورود</th>
              <th className="p-2">قیمت خروج</th>
              <th className="p-2">دلیل خروج</th>
              <th className="p-2">سود خالص</th>
              <th className="p-2 text-center">%</th>
              <th className="p-2 text-center">مدت</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {sorted.map((t) => (
              <tr
                key={t.id}
                className="hover:bg-gray-50 dark:hover:bg-gray-700/30"
              >
                <td className="p-2 font-mono">{t.isin}</td>
                <td className="p-2">{fmtDate(t.entryTime)}</td>
                <td className="p-2">{fmtDate(t.exitTime)}</td>
                <td className="p-2 text-center">{fmtMoney(t.totalEntryQuantity)}</td>
                <td className="p-2">{fmtMoney(t.weightedAvgEntryPrice)}</td>
                <td className="p-2">{fmtMoney(t.exitPrice)}</td>
                <td className="p-2">
                  <ExitBadge reason={t.exitReason} />
                </td>
                <td
                  className={`p-2 font-mono ${
                    t.netPnL > 0 ? "text-emerald-600" : "text-red-600"
                  }`}
                >
                  {fmtMoney(t.netPnL)}
                </td>
                <td
                  className={`p-2 text-center ${
                    t.netPnLPct > 0 ? "text-emerald-600" : "text-red-600"
                  }`}
                >
                  {t.netPnLPct.toFixed(2)}%
                </td>
                <td className="p-2 text-center text-gray-500">
                  {fmtDuration(t.holdingDurationSeconds)}
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

    <td className="p-2 font-medium text-gray-600 dark:text-gray-400">
      {label}
    </td>
    {values.map((v, idx) => (
      <td
        key={idx}
        className={`p-2 text-center font-mono ${
          bestIndex === idx
            ? "bg-emerald-50 text-emerald-700 font-bold dark:bg-emerald-900/20 dark:text-emerald-300"
            : ""
        }`}
      >
        {typeof v === "number" && format ? format(v) : v}
      </td>
    ))}
  </tr>
);

const findBestIndex = (
  values: number[],
  highIsBetter: boolean,
): number | null => {
  if (values.length === 0) return null;
  let best = 0;
  for (let i = 1; i < values.length; i++) {
    if (highIsBetter ? values[i] > values[best] : values[i] < values[best]) {
      best = i;
    }
  }
  return best;
};

//src/modules/Trader/pages/Backtests/BacktestDetailPage.tsx
import React, { useEffect, useState } from "react";
import { Link, useParams } from "react-router-dom";
import { backtestApi } from "../../api/backtestApi";
import type {
  BacktestRunDetailView,
  BacktestTradeView,
  BacktestDailyEquityView,
} from "../../models";
import { BacktestStatus, BACKTEST_STATUS_LABELS } from "../../models";
import { BacktestStatusBadge } from "./components/BacktestStatusBadge";
import { MetricsCards } from "./components/MetricsCards";
import { EquityCurveChart } from "./components/EquityCurveChart";
import { TradesTable } from "./components/TradesTable";
import { BacktestLogsTab } from "./components/BacktestLogsTab";
export const BacktestDetailPage: React.FC = () => {
  const { id } = useParams<{ id: string }>();
  const [run, setRun] = useState<BacktestRunDetailView | null>(null);
  const [trades, setTrades] = useState<BacktestTradeView[]>([]);
  const [equity, setEquity] = useState<BacktestDailyEquityView[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
const [activeTab, setActiveTab] = useState<"summary" | "logs">("summary");
  const load = async () => {
    if (!id) return;
    setLoading(true);
    setError(null);
    try {
      const detail = await backtestApi.GetById(id);
      setRun(detail);

      if (detail.status === BacktestStatus.Completed) {
        const [t, e] = await Promise.all([
          backtestApi.GetTrades(id),
          backtestApi.GetEquityCurve(id),
        ]);
        setTrades(t);
        setEquity(e);
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : "خطا در دریافت");
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    void load();
  }, [id]);

  // Poll if running
  useEffect(() => {
    if (!run) return;
    if (
      run.status !== BacktestStatus.Pending &&
      run.status !== BacktestStatus.Running
    )
      return;

    const t = setInterval(() => {
      void load();
    }, 5000);

    return () => clearInterval(t);
  }, [run?.status, id]);

  if (loading && !run) {
    return (
      <div className="p-6 text-center text-sm text-gray-500">
        در حال بارگذاری...
      </div>
    );
  }

  if (error || !run) {
    return (
      <div className="p-6">
        <div className="rounded-lg bg-red-50 p-4 text-sm text-red-700 dark:bg-red-900/30 dark:text-red-400">
          ⚠️ {error ?? "بکتست پیدا نشد"}
        </div>
        <Link
          to="/trader/backtests"
          className="mt-4 inline-block text-blue-600 hover:underline"
        >
          ← بازگشت
        </Link>
      </div>
    );
  }

  return (
    <div className="space-y-4 p-6">
      <div className="flex items-center justify-between">
        <div>
          <Link
            to="/trader/backtests"
            className="text-xs text-blue-600 hover:underline"
          >
            ← بکتست‌ها
          </Link>
          <h1 className="mt-1 text-xl font-bold text-gray-900 dark:text-white">
            {run.name}
          </h1>
          <p className="mt-1 text-xs text-gray-500">
            {run.strategyKey} · {run.dateFrom.slice(0, 10)} →{" "}
            {run.dateTo.slice(0, 10)} ·{" "}
            {run.symbols.map((s) => s.isin).join(", ")}
          </p>
        </div>
        <div className="text-right">
          <BacktestStatusBadge status={run.status} />
          {run.durationMs && (
            <p className="mt-1 text-[10px] text-gray-500">
              {(run.durationMs / 1000).toFixed(1)}s
            </p>
          )}
        </div>
      </div>

      {run.errorMessage && (
        <div className="rounded-lg bg-red-50 p-3 text-sm text-red-700 dark:bg-red-900/30 dark:text-red-400">
          ⚠️ {run.errorMessage}
        </div>
      )}
{/* Tab Bar */}
<div className="flex gap-1 border-b border-gray-200 dark:border-gray-700">
  <button
    type="button"
    onClick={() => setActiveTab("summary")}
    className={`border-b-2 px-4 py-2 text-sm font-medium transition-colors ${
      activeTab === "summary"
        ? "border-blue-500 text-blue-600 dark:text-blue-400"
        : "border-transparent text-gray-500 hover:text-gray-700 dark:text-gray-400"
    }`}
  >
    خلاصه
  </button>
  <button
    type="button"
    onClick={() => setActiveTab("logs")}
    className={`border-b-2 px-4 py-2 text-sm font-medium transition-colors ${
      activeTab === "logs"
        ? "border-blue-500 text-blue-600 dark:text-blue-400"
        : "border-transparent text-gray-500 hover:text-gray-700 dark:text-gray-400"
    }`}
  >
    لاگ‌ها
  </button>
</div>
{activeTab === "summary" && (
  <>
    {run.status === BacktestStatus.Completed && (
      <>
        <MetricsCards run={run} />
        <EquityCurveChart
          series={[
            {
              name: run.name,
              data: equity,
              color: "#10b981",
            },
          ]}
        />
        <TradesTable trades={trades} />
      </>
    )}

    {run.status === BacktestStatus.Pending && (
      <div className="rounded-lg border border-blue-700 bg-blue-900/20 p-4 text-sm text-blue-300">
        ⏳ در انتظار اجرا...
      </div>
    )}

    {run.status === BacktestStatus.Running && (
      <div className="rounded-lg border border-amber-700 bg-amber-900/20 p-4 text-sm text-amber-300">
        ⚡ در حال اجرا...
      </div>
    )}
  </>
)}

{activeTab === "logs" && <BacktestLogsTab backtestRunId={run.id} />}
      

      

    </div>
  );
};


//src/modules/Trader/pages/Backtests/BacktestsListPage.tsx
import React, { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import { useBacktestManagement } from "../../hooks/useBacktestManagement";
import { BacktestStatusBadge } from "./components/BacktestStatusBadge";
import { BacktestCreateModal } from "./BacktestCreateModal";
import { BacktestStatus } from "../../models";
import { useNavigate } from "react-router-dom";
const fmtMoney = (v: number): string =>
  v.toLocaleString("fa-IR", { maximumFractionDigits: 0 });

const fmtPct = (v: number): string => `${v.toFixed(2)}%`;

export const BacktestsListPage: React.FC = () => {
  const { runs, loading, error, refresh, handleCancel } =
    useBacktestManagement();
  const [createOpen, setCreateOpen] = useState(false);
const navigate = useNavigate();
const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());

const toggleId = (id: string) => {
  setSelectedIds((prev) => {
    const next = new Set(prev);
    if (next.has(id)) next.delete(id);
    else next.add(id);
    return next;
  });
};

const clearSelection = () => setSelectedIds(new Set());

const handleCompare = () => {
  if (selectedIds.size < 2) return;
  navigate(
    `/trader/backtests/compare?ids=${Array.from(selectedIds).join(",")}`,
  );
};
  // Poll running backtests
  useEffect(() => {
    const hasRunning = runs.some(
      (r) =>
        r.status === BacktestStatus.Pending ||
        r.status === BacktestStatus.Running,
    );
    if (!hasRunning) return;

    const t = setInterval(() => {
      void refresh();
    }, 5000);

    return () => clearInterval(t);
  }, [runs, refresh]);

  return (
    <div className="space-y-4 p-6">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-xl font-bold text-gray-900 dark:text-white">
            بکتست‌ها
          </h1>
          <p className="mt-1 text-xs text-gray-500 dark:text-gray-400">
            {runs.length} بکتست
          </p>
        </div>
        <div className="flex gap-2">
          <button
            type="button"
            onClick={refresh}
            disabled={loading}
            className="rounded-lg border border-gray-300 px-3 py-2 text-sm text-gray-700 hover:bg-gray-100 disabled:opacity-50 dark:border-gray-600 dark:text-gray-300"
          >
            🔄 بروزرسانی
          </button>
          {selectedIds.size >= 2 && (
  <button
    type="button"
    onClick={handleCompare}
    className="rounded-lg bg-emerald-600 px-4 py-2 text-sm font-medium text-white hover:bg-emerald-700"
  >
    📊 مقایسه ({selectedIds.size})
  </button>
)}
          <button
            type="button"
            onClick={() => setCreateOpen(true)}
            className="rounded-lg bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700"
          >
            ➕ بکتست جدید
          </button>
        </div>
      </div>

      {error && (
        <div className="rounded-lg bg-red-50 p-3 text-sm text-red-700 dark:bg-red-900/30 dark:text-red-400">
          ⚠️ {error}
        </div>
      )}

      <div className="overflow-x-auto rounded-lg border border-gray-200 shadow-sm dark:border-gray-700">
        <table className="w-full text-right text-sm text-gray-700 dark:text-gray-300">
          <thead className="bg-gray-50 text-xs text-gray-700 dark:bg-gray-700 dark:text-gray-300">
           <tr>
  <th className="w-10 p-3">
    {selectedIds.size > 0 && (
      <button
        type="button"
        onClick={clearSelection}
        className="text-xs text-red-600 hover:underline"
        title="پاک کردن انتخاب"
      >
        ✕
      </button>
    )}
  </th>
              <th className="p-3">نام</th>
              <th className="p-3">وضعیت</th>
              <th className="p-3">استراتژی</th>
              <th className="p-3">بازه</th>
              <th className="p-3">بازده</th>
              <th className="p-3">حداکثر افت</th>
              <th className="p-3">شارپ</th>
              <th className="p-3">ترید</th>
              <th className="p-3 text-center">عملیات</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-200 dark:divide-gray-700">
            {loading && runs.length === 0 ? (
              <tr>
                <td colSpan={10} className="p-6 text-center">
                  در حال بارگذاری...
                </td>
              </tr>
            ) : runs.length === 0 ? (
              <tr>
                <td colSpan={10} className="p-6 text-center text-gray-500">
                  هیچ بکتستی وجود ندارد.
                </td>
              </tr>
            ) : (
              runs.map((r) => (
  <tr
    key={r.id}
    className={`hover:bg-gray-50 dark:hover:bg-gray-700/30 ${
      selectedIds.has(r.id) ? "bg-blue-50 dark:bg-blue-900/20" : ""
    }`}
  >
    <td className="p-3">
      <input
        type="checkbox"
        checked={selectedIds.has(r.id)}
        onChange={() => toggleId(r.id)}
        className="rounded"
      />
    </td>
    <td className="p-3 font-medium">{r.name}</td>
                  <td className="p-3">
                    <BacktestStatusBadge status={r.status} />
                  </td>
                  <td className="p-3 font-mono text-xs">{r.strategyKey}</td>
                  <td className="p-3 text-xs">
                    {r.dateFrom.slice(0, 10)} → {r.dateTo.slice(0, 10)}
                  </td>
                  <td
                    className={`p-3 text-xs font-mono ${
                      r.totalReturnPct > 0
                        ? "text-emerald-600"
                        : r.totalReturnPct < 0
                          ? "text-red-600"
                          : ""
                    }`}
                  >
                    {fmtPct(r.totalReturnPct)}
                  </td>
                  <td className="p-3 text-xs font-mono text-red-600">
                    {fmtPct(-Math.abs(r.maxDrawdownPct))}
                  </td>
                  <td className="p-3 text-xs font-mono">
                    {r.sharpeRatio.toFixed(2)}
                  </td>
                  <td className="p-3 text-xs">
                    {r.totalTrades} ({r.winningTrades}W/{r.losingTrades}L)
                  </td>
                  <td className="p-3">
                    <div className="flex justify-end gap-1">
                      <Link
                        to={`/trader/backtests/${r.id}`}
                        className="rounded bg-blue-600 px-2 py-1 text-xs text-white hover:bg-blue-700"
                      >
                        جزئیات
                      </Link>
                      {(r.status === BacktestStatus.Pending ||
                        r.status === BacktestStatus.Running) && (
                        <button
                          type="button"
                          onClick={() => handleCancel(r.id)}
                          className="rounded bg-red-600 px-2 py-1 text-xs text-white hover:bg-red-700"
                        >
                          لغو
                        </button>
                      )}
                    </div>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>

      <BacktestCreateModal
        isOpen={createOpen}
        onClose={() => setCreateOpen(false)}
        onCreated={() => refresh()}
      />
    </div>
  );
};
